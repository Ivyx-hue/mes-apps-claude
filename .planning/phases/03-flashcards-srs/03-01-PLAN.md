// Loaded via <script src="srs.js" defer> in the browser, or `require('./srs.js')` in Node.
// Idempotent double-load guard (mirrors outils-data.js WR-04).
window.SRS = {
  // Pure function. row = current per-card row OR null (new card); grade ∈ GRADE; today = 'yyyy-mm-dd'.
  // Returns a NEW row object (does NOT mutate input).
  schedule(row, grade, today): { ease, interval, due, lapses, reps, introduced },

  // Pure boolean. row.due <= today via lexicographic yyyy-mm-dd compare.
  isDue(row, today): boolean,

  // Pure. Parse 'yyyy-mm-dd' as LOCAL via (year, monthIndex, day) constructor; add n days; reformat.
  addDays(yyyymmdd, n): 'yyyy-mm-dd',

  // Pure. Returns new Date().toLocaleDateString('sv-SE') — user's local civil day.
  todayLocal(): 'yyyy-mm-dd',

  // Pure helper. Returns array of cards from `cards` where `store[card.id]` exists AND row.due <= today.
  filterDue(cards, store, today): Card[],

  // Pure helper. Returns count of store entries where row.introduced === today.
  countNew(store, today): number,

  // Object.freeze'd constants from D-07.
  DEFAULTS: {
    EASE_INIT: 2.5,
    EASE_FLOOR: 1.3,
    FIRST_INTERVAL: 1,
    SECOND_INTERVAL: 6,
    RATE_RESET_INTERVAL: 1,
    RATE_EASE_DELTA: -0.20,
    DUR_INTERVAL_MULT: 1.2,
    DUR_EASE_DELTA: -0.15,
    BIEN_EASE_DELTA: 0,
    FACILE_BONUS_MULT: 1.3,
    FACILE_EASE_DELTA: 0.15
  },

  // Object.freeze'd grade enum.
  GRADE: { RATE: 'rate', DUR: 'dur', BIEN: 'bien', FACILE: 'facile' }
};
